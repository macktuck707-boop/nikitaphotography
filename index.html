    <script>

        let currentLbIndex = 1;

        let currentLbPrefix = '';

        let currentLbCount = 0;


        function createGrid(containerId, prefix, count) {

            const grid = document.getElementById(containerId);

            for (let i = 1; i <= count; i++) {

                const src = `photos/${i}${prefix}.jpg`;

                const img = document.createElement('img');

                img.src = src;

                img.onclick = () => openLightbox(i, prefix, count);

                grid.appendChild(img);

            }

        }


        // Подключение подгрузки фото:

        createGrid('grid-l', 'l', 31); // Пейзажи (1l.jpg - 31l.jpg)

        createGrid('grid-a', 'a', 14); // Животные (1a.jpg - 14a.jpg)

        createGrid('grid-f', 'f', 18); // Фестивали (1f.jpg - 18f.jpg)

        createGrid('grid-p', 'p', 10); // Портфолио (1p.jpg - 10p.jpg) — поменяй число 10 на нужное количество файлов!


        function openLightbox(index, prefix, count) {

            currentLbIndex = index;

            currentLbPrefix = prefix;

            currentLbCount = count;

            

            const lb = document.getElementById('lightbox');

            const lbImg = document.getElementById('lightbox-img');

            const thumbs = document.getElementById('lb-thumbs');

            

            const src = `photos/${index}${prefix}.jpg`;

            lbImg.src = src;

            

            thumbs.innerHTML = '';

            for (let i = 1; i <= count; i++) {

                const tSrc = `photos/${i}${prefix}.jpg`;

                const thumb = document.createElement('img');

                thumb.src = tSrc;

                if(i === index) thumb.className = 'active';

                thumb.onclick = (e) => { e.stopPropagation(); openLightbox(i, prefix, count); };

                thumbs.appendChild(thumb);

            }

            lb.classList.add('active');


            const activeThumb = thumbs.querySelector('.active');

            if (activeThumb) {

                activeThumb.scrollIntoView({ behavior: 'smooth', inline: 'center', block: 'nearest' });

            }

        }


        function changeImage(step, event) {

            if (event) event.stopPropagation();

            

            let newIndex = currentLbIndex + step;

            

            if (newIndex > currentLbCount) newIndex = 1;

            if (newIndex < 1) newIndex = currentLbCount;

            

            openLightbox(newIndex, currentLbPrefix, currentLbCount);

        }


        function closeLightbox() { document.getElementById('lightbox').classList.remove('active'); }

        

        document.getElementById('lightbox').addEventListener('click', (e) => {

            if (e.target.id === 'lightbox') {

                closeLightbox();

            }

        });


        document.addEventListener('keydown', (e) => {

            const lb = document.getElementById('lightbox');

            if (lb.classList.contains('active')) {

                if (e.key === 'ArrowLeft') changeImage(-1);

                else if (e.key === 'ArrowRight') changeImage(1);

                else if (e.key === 'Escape') closeLightbox();

            }

        });


        function showGallery(id) {

            document.getElementById('main').style.display = 'none';

            document.getElementById(id).classList.add('active');

            window.scrollTo(0,0);

        }


        function hideGallery() {

            document.getElementById('main').style.display = 'flex';

        }
